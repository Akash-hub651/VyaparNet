## PART 1: API CONTRACTS

### 1. API STANDARDS & PHILOSOPHY

#### Response Envelope (Standard for ALL APIs)
```typescript
// SUCCESS
interface ApiResponse<T> {
  success: true;
  message: string;
  data: T;
  meta?: PaginationMeta | Record<string, any>;
  traceId: string;
}

// ERROR
interface ApiErrorResponse {
  success: false;
  error: {
    code: string;        // Machine-readable error code
    message: string;     // Human-readable message
    details?: Record<string, any>;
    timestamp: string;
  };
  traceId: string;
  requestId: string;
}

interface PaginationMeta {
  total: number;
  page: number;
  limit: number;
  totalPages: number;
  hasNextPage: boolean;
  hasPrevPage: boolean;
}
```

#### HTTP Standards
| Standard | Rule |
|----------|------|
| **Base Path** | `/api/v1/` |
| **Content-Type** | `application/json` |
| **Authentication** | `Authorization: Bearer <token>` |
| **Idempotency** | `Idempotency-Key: <uuid>` for POST/PUT |
| **Language** | `Accept-Language: hi-en` |
| **Request ID** | `X-Request-Id: <uuid>` (auto-generated if missing) |

#### Pagination (Cursor-Based for Lists)
```typescript
// Request
interface PaginationParams {
  cursor?: string;      // Optional: offset cursor
  limit?: number;       // Default: 20, Max: 100
  sort?: string;        // e.g., "createdAt:desc" or "price:asc"
}

// Response Meta
interface CursorPaginationMeta {
  nextCursor: string | null;
  prevCursor: string | null;
  limit: number;
  total?: number;       // Optional: expensive to calculate on large tables
}
```

### 2. AUTHENTICATION & AUTHORIZATION CONTRACTS

#### Authentication Flow
```
[Step 1] POST /api/v1/auth/otp/send
         Body: { phoneNumber: "+919876543210" }
         Response: { success: true, message: "OTP sent", data: { expiresIn: 300 } }

[Step 2] POST /api/v1/auth/otp/verify
         Body: { phoneNumber: "+919876543210", otp: "123456", deviceId?: "dev-abc" }
         Response: { success: true, data: { accessToken, refreshToken, user } }

[Step 3] [Authenticated Request]
         Header: Authorization: Bearer <accessToken>

[Step 4] GET /api/v1/auth/refresh
         Header: Authorization: Bearer <refreshToken>
         Response: { success: true, data: { accessToken, refreshToken } }
```

### 13. AUTHENTICATION & RBAC IMPLEMENTATION

#### JWT Guard
```typescript
// guards/jwt-auth.guard.ts
@Injectable()
export class JwtAuthGuard implements CanActivate {
  constructor(private jwtService: JwtService, private userService: UserService) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request = context.switchToHttp().getRequest();
    const token = this.extractToken(request);
    
    if (!token) throw new UnauthorizedException();
    
    const payload = this.jwtService.verify(token);
    request.user = await this.userService.findById(payload.sub);
    return true;
  }
}
```

#### RBAC Decorator & Guard
```typescript
// decorators/roles.decorator.ts
export const ROLES_KEY = 'roles';
export const Roles = (...roles: Role[]) => SetMetadata(ROLES_KEY, roles);

// decorators/permissions.decorator.ts
export const PERMISSIONS_KEY = 'permissions';
export const Permissions = (...permissions: Permission[]) => SetMetadata(PERMISSIONS_KEY, permissions);

// guards/roles.guard.ts
@Injectable()
export class RolesGuard implements CanActivate {
  constructor(private reflector: Reflector) {}

  canActivate(context: ExecutionContext): boolean {
    const requiredRoles = this.reflector.getAllAndOverride<Role[]>(ROLES_KEY, [
      context.getHandler(), context.getClass(),
    ]);
    if (!requiredRoles) return true;
    
    const { user } = context.switchToHttp().getRequest();
    return requiredRoles.some((role) => user.role === role);
  }
}
```

### 19.3 Feature Flag Middleware Flow

```typescript
// core/feature-flags/feature-flag.middleware.ts
@Injectable()
export class FeatureFlagMiddleware implements NestMiddleware {
  private readonly localCache = new Map<string, { value: boolean; ts: number }>();
  private readonly LOCAL_TTL = 30 * 1000; // 30 seconds local

  async use(req: Request, res: Response, next: NextFunction) {
    const flags = await this.getFlagsForUser(req.user?.id);
    req['featureFlags'] = flags;
    next();
  }

  private async getFlagsForUser(userId?: string): Promise<Record<string, boolean>> {
    const cacheKey = `flags:${userId || 'anon'}`;
    
    // 1. Local memory (fastest)
    const local = this.localCache.get(cacheKey);
    if (local && Date.now() - local.ts < this.LOCAL_TTL) return local.value;

    // 2. Redis lookup
    const redisValue = await this.redis.get(cacheKey);
    const parsed = redisValue ? JSON.parse(redisValue) : await this.fetchFromDb(userId);
    
    this.localCache.set(cacheKey, { value: parsed, ts: Date.now() });
    return parsed;
  }
}

// Usage in controller
@Controller('search')
export class SearchController {
  constructor(private readonly searchService: SearchService) {}

    if (req['featureFlags']?.['search-v2']) {
      return this.searchService.searchV2(query);
    }
    return this.searchService.searchV1(query);
  }
}
```

### 20.6 Webhook Retry Strategy

```typescript
// BullMQ worker for webhooks
interface WebhookJob {
  url: string;
  payload: any;
  eventType: string;
  attempt: number;
  maxAttempts: number;
  // Retry: 2s, 4s, 8s, 16s, 32s (exponential)
}

@Processor(WEBHOOK_QUEUE)
export class WebhookWorker extends WorkerHost {
  @Process('send')
  async handleWebhook(job: Job<WebhookJob>) {
    const { url, payload, eventType, attempt, maxAttempts } = job.data;
    
    try {
      const response = await axios.post(url, payload, {
        headers: { 'X-Webhook-Signature': this.generateHmac(payload) },
        timeout: 8000,
      });
      return { success: true, status: response.status };
    } catch (error) {
      if (attempt < maxAttempts) {
        throw new WebhookRetryError(`Retry ${attempt}/${maxAttempts}`, error);
      }
      // Last attempt → DLQ
      await job.moveToFailed(error, { removeOnFail: false });
      throw error;
    }
  }
}
```

```typescript
// Worker process
@Processor(SEARCH_QUEUE)
export class SearchReindexWorker extends WorkerHost {
  @Process('reindex-all')
  async reindexAll() {
    // 1. Lock search writes
    await this.redis.set('search:lock', 'reindexing', 'EX', 3600);
    
    // 2. Rebuild index in temp table
    const products = await this.productRepository.findAll();
    for (const product of products) {
      await this.searchService.upsertTemp(product);
    }
    
    // 3. Atomic swap (rename temp → main)
    await this.searchService.swapIndex();
    
    // 4. Unlock
    await this.redis.del('search:lock');
  }
}
```
